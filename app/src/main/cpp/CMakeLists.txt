cmake_minimum_required(VERSION 3.22)
project(XWAQuestM1 LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "M1 requires Android arm64-v8a")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SDL_ROOT}" rev-parse HEAD OUTPUT_VARIABLE SDL_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SDL_REV STREQUAL "8f8ed757bc94d2e097aab8c4c0f58b57dcee9871")
    message(FATAL_ERROR "Unexpected SDL revision: ${SDL_REV}")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_GPU_OPENXR ON CACHE BOOL "" FORCE)
set(SDL_VULKAN ON CACHE BOOL "" FORCE)
add_subdirectory("${SDL_ROOT}" "${CMAKE_BINARY_DIR}/SDL3")
add_library(main SHARED main.c)
target_compile_features(main PRIVATE c_std_11)
target_compile_definitions(main PRIVATE HAVE_OPENXR_H=1 XWAQUEST_SDL_REV="${SDL_REV}")
target_include_directories(main PRIVATE "${SDL_ROOT}/src/video/khronos" "${SDL_ROOT}/test")
target_link_libraries(main PRIVATE SDL3::SDL3 log)
